cmake_minimum_required(VERSION 3.16)
project(airxonix_native LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(AIRXONIX_BUILD_RUNTIME "Build the SDL2/OpenGL ES runtime" ON)
option(AIRXONIX_BUILD_TESTS "Build gameplay-only reverse-engineering tests" OFF)
option(AIRXONIX_PORTMASTER_ARM64_COMPAT "Force ARMv8-A/Cortex-A53 ISA for H700/RK3326 PortMaster builds" OFF)
option(AIRXONIX_RESOURCE_FREE "Disable loose-PNG decoding and use built-in replacement resources" OFF)

# The PortMaster Universal GCC9 toolchain itself defaults to ARMv8.2-A and
# Cortex-A75/A55. H700 and RK3326 devices are Cortex-A53-class ARMv8-A, so a
# cross-build must override the toolchain default for every translation unit.
if(AIRXONIX_PORTMASTER_ARM64_COMPAT)
    add_compile_options(-march=armv8-a -mcpu=cortex-a53 -mtune=cortex-a53)
endif()

add_library(airxonix_gamecore STATIC ${AIRXONIX_GAMECORE_SOURCES})
target_include_directories(airxonix_gamecore PUBLIC src)
# Fidelity matters more than an unmeasurable fast-math gain on H700.  In
# particular, grid collision uses round-to-nearest and reverse-engineered
# floating thresholds; -ffast-math may legally change those semantics.
target_compile_options(airxonix_gamecore PRIVATE
    -O2 -fno-math-errno -fno-trapping-math -ffp-contract=off
    -fomit-frame-pointer -Wall -Wextra -Wpedantic)

if(AIRXONIX_BUILD_RUNTIME)
    if(ANDROID)
        # SDL is built from the matching official SDL2 source by the app's
        # outer CMakeLists.txt. No Linux pkg-config or libpng on Android.
        set(AIRXONIX_RESOURCE_FREE ON)
        set(_AX_RUNTIME_TARGET main)
        add_library(main SHARED
            src/main.cpp
            src/wavepack.cpp
            src/core/paths.cpp
            src/core/paths_discovery.cpp
            src/core/legacy_clock.cpp
            src/platform/input.cpp
            src/audio/music.cpp
            src/audio/legacy_sfx_bank.cpp
            src/render/renderer.cpp
            src/render/legacy_atlas_runtime.cpp
        )
        target_include_directories(main PRIVATE src)
        target_compile_definitions(main PRIVATE AIRXONIX_NO_PNG)
        target_compile_options(main PRIVATE -O2 -fno-math-errno -fno-trapping-math
          -ffp-contract=off -fomit-frame-pointer -Wall -Wextra -Wpedantic)
        target_link_libraries(main PRIVATE airxonix_gamecore airxonix_legacy_mesh
          SDL2 SDL2main GLESv2 log android m)
    else()
    if(AIRXONIX_ARM64_SDK_ROOT AND AIRXONIX_ARM64_LINK_DIR)
        set(SDL2_INCLUDE_DIRS "${AIRXONIX_ARM64_SDK_ROOT}/sysroot/usr/include/SDL2")
        set(GLES2_INCLUDE_DIRS "${AIRXONIX_ARM64_SDK_ROOT}/sysroot/usr/include/SDL2")
        set(SDL2_LIBRARIES "${AIRXONIX_ARM64_LINK_DIR}/libSDL2.so")
        set(GLES2_LIBRARIES "${AIRXONIX_ARM64_LINK_DIR}/libGLESv2.so")
        set(SDL2_LIBRARY_DIRS "${AIRXONIX_ARM64_LINK_DIR}")
        set(GLES2_LIBRARY_DIRS "${AIRXONIX_ARM64_LINK_DIR}")
        if(NOT AIRXONIX_RESOURCE_FREE)
            message(FATAL_ERROR "ARM64 build requires AIRXONIX_RESOURCE_FREE=ON")
        endif()
    elseif(AIRXONIX_ARMHF_SDK_ROOT AND AIRXONIX_ARMHF_LINK_DIR)
        # H700 ARMHF cross build. Runtime libs are resolved by PortMaster on
        # the target; generated OpenGL ES link stubs must never be distributed.
        set(SDL2_INCLUDE_DIRS "${AIRXONIX_ARMHF_SDK_ROOT}/include/SDL2")
        set(GLES2_INCLUDE_DIRS "${AIRXONIX_ARMHF_SDK_ROOT}/compat_include")
        set(SDL2_LIBRARIES "${AIRXONIX_ARMHF_LINK_DIR}/libSDL2.so")
        set(GLES2_LIBRARIES "${AIRXONIX_ARMHF_LINK_DIR}/libGLESv2.so")
        set(SDL2_LIBRARY_DIRS "${AIRXONIX_ARMHF_LINK_DIR}")
        set(GLES2_LIBRARY_DIRS "${AIRXONIX_ARMHF_LINK_DIR}")
        if(NOT AIRXONIX_RESOURCE_FREE)
            message(FATAL_ERROR "ARMHF build requires AIRXONIX_RESOURCE_FREE=ON")
        endif()
    else()
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(SDL2 REQUIRED sdl2)
        pkg_check_modules(GLES2 REQUIRED glesv2)
        if(NOT AIRXONIX_RESOURCE_FREE)
            pkg_check_modules(PNG REQUIRED libpng)
        endif()
    endif()

    add_executable(airxonix
        src/main.cpp
        src/net/lan_transport.cpp
        src/net/lan_discovery.cpp
        src/wavepack.cpp
        src/core/paths.cpp
        src/core/paths_discovery.cpp
        src/core/legacy_clock.cpp
        src/platform/input.cpp
        src/audio/music.cpp
        src/audio/legacy_sfx_bank.cpp
        src/render/renderer.cpp
        src/render/legacy_atlas_runtime.cpp
    )
    target_include_directories(airxonix PRIVATE ${SDL2_INCLUDE_DIRS} ${GLES2_INCLUDE_DIRS} src)
    target_compile_options(airxonix PRIVATE
        ${SDL2_CFLAGS_OTHER} ${GLES2_CFLAGS_OTHER}
        -O2 -fno-math-errno -fno-trapping-math -ffp-contract=off
        -fomit-frame-pointer -Wall -Wextra -Wpedantic)
    target_link_directories(airxonix PRIVATE ${SDL2_LIBRARY_DIRS} ${GLES2_LIBRARY_DIRS})
    target_link_libraries(airxonix PRIVATE airxonix_gamecore airxonix_legacy_mesh ${SDL2_LIBRARIES} ${GLES2_LIBRARIES} m dl pthread)
    if(AIRXONIX_ARM64_SDK_ROOT AND AIRXONIX_ARM64_LINK_DIR)
        target_link_libraries(airxonix PRIVATE stdc++ gcc_s c
            "${AIRXONIX_ARM64_SDK_ROOT}/sysroot/usr/lib/crtn.o")
    endif()
    if(AIRXONIX_RESOURCE_FREE)
        target_compile_definitions(airxonix PRIVATE AIRXONIX_NO_PNG)
    else()
        target_include_directories(airxonix PRIVATE ${PNG_INCLUDE_DIRS})
        target_compile_options(airxonix PRIVATE ${PNG_CFLAGS_OTHER})
        target_link_directories(airxonix PRIVATE ${PNG_LIBRARY_DIRS})
        target_link_libraries(airxonix PRIVATE ${PNG_LIBRARIES})
    endif()
    endif() # non-Android SDL2 pkg-config runtime
endif()
